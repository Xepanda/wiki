# 路线 2 · PC 性能与 Git 工作流实时监控副屏

将 Kindle 摆在电脑显示器旁，作为一块**无频闪、不伤眼的极客硬件监控副屏 (Workstation HUD)**。

---


* **常亮保障**：在调试前执行 `ssh kindle "lipc-set-prop com.lab126.powerd preventScreenSaver 1"`，确保设备永不休眠；
* **局部刷新**：使用 `eips -g /mnt/us/hud.png` 执行局部快速更新，避免全屏黑白闪烁，观感极为舒适。

## 2. 核心监控脚本 (`monitor_hud.py`)

```python
#!/usr/bin/env python3
# -*- coding: utf-8 -*-
"""
PC Performance & Git Status HUD Monitor for Kindle 8
"""
import os, sys, subprocess, psutil
from datetime import datetime
from PIL import Image, ImageDraw, ImageFont

WIDTH, HEIGHT = 800, 600

def get_git_status(repo_path="D:\\ePanda\\wiki"):
    branch = subprocess.run(['git', 'branch', '--show-current'], cwd=repo_path, capture_output=True, text=True).stdout.strip()
    status = subprocess.run(['git', 'status', '--porcelain'], cwd=repo_path, capture_output=True, text=True).stdout.strip()
    return branch or 'master', len(status.splitlines()) if status else 0

def render_hud(output_path='hud.png'):
    img = Image.new('L', (WIDTH, HEIGHT), color=255)
    draw = ImageDraw.Draw(img)
    font_path = "C:\\Windows\\Fonts\\msyh.ttc"

    cpu_percent = psutil.cpu_percent(interval=0.5)
    ram = psutil.virtual_memory()
    disk_c = psutil.disk_usage('C:\\')
    branch, git_mods = get_git_status()

    # 1. 顶部 Header
    draw.rectangle([(20, 15), (780, 65)], fill=240, outline=0, width=2)
    draw.text((35, 25), "⚡ PC WORKSTATION HUD · 性能与工作流监控屏", fill=0, font=ImageFont.truetype(font_path, 24))
    draw.text((630, 28), datetime.now().strftime("%H:%M:%S"), fill=0, font=ImageFont.truetype(font_path, 18))

    # 2. CPU & RAM 进度卡片
    draw.rectangle([(20, 80), (390, 200)], fill=255, outline=0, width=2)
    draw.text((35, 92), "💻 CPU 利用率", fill=60, font=ImageFont.truetype(font_path, 15))
    draw.text((35, 115), f"{cpu_percent:.1f}%", fill=0, font=ImageFont.truetype(font_path, 34))

    draw.rectangle([(410, 80), (780, 200)], fill=255, outline=0, width=2)
    draw.text((425, 92), "🧠 物理内存 (RAM)", fill=60, font=ImageFont.truetype(font_path, 15))
    draw.text((425, 115), f"{ram.percent:.1f}%", fill=0, font=ImageFont.truetype(font_path, 34))

    # 3. 磁盘与 Git 状态
    draw.rectangle([(410, 215), (780, 335)], fill=255, outline=0, width=2)
    draw.text((425, 227), "🌿 活跃 Git 仓库 (PandaWiki)", fill=60, font=ImageFont.truetype(font_path, 15))
    draw.text((425, 255), f"• 当前分支: {branch}", fill=0, font=ImageFont.truetype(font_path, 18))
    draw.text((425, 290), f"• 待提交变更: {git_mods} files", fill=0, font=ImageFont.truetype(font_path, 18))

    img.save(output_path)
    return output_path

def push_to_kindle(img_path='hud.png'):
    subprocess.run(f'scp -o ConnectTimeout=4 "{img_path}" root@kindle:/mnt/us/hud.png', shell=True)
    subprocess.run('ssh kindle "eips -g /mnt/us/hud.png"', shell=True)

if __name__ == '__main__':
    push_to_kindle(render_hud())
```
